  // ---------------------------------------------------------------- BLOCK (2026-11-10, the user's design for every character)
  // "Blocked hit cannot crit and block reduces the damage taken by 20% (may change in the future)." A block is a REACTION like the Dodge (the user, 2026-11-10: it happens when a hit lands, not a stance): the blocker's block
  // pack plays so that its guard is up when the blow lands (placement idx: the block frame), the damage is cut by BLOCK_REDUCE, a blocked hit is never a critical one, and the blow meets the MAGE'S OWN BARRIER: brDraw, the same
  // renderer as her passive (the half circle of glass on the side the blow comes from, the glow on the ground, the ripple where the blow lands, the flash), on any fighter, short: it snaps up in BLOCK_CAST_MS, holds
  // BLOCK_HOLD_MS and lets go over BLOCK_DROP_MS (`L.geo` in brState / brDraw gives it the blocker's size and place instead of the mage's). (The first version, a few hours earlier, was a NEW drawing in the same colours; the user
  // asked whether it was her Barrier and said it went below the feet: it is her Barrier now, and with `geo` the circle's bottom is ON the ground line (the clip stops there) and the ground ring is flattened.) No white hit flash and
  // no hit pose for a blocked blow, only a small recoil and a faint brighten. The page only DRAWS it: the numbers are the engine's (docs/SKILL_BEHAVIOR_LATER.md). A turn is blocked with `block: true` in the script (the scenario
  // 'the monster blocks two strikes'). Wired so far (the blocker is a monster that has `MONSTERS[key].block`): the warrior's attack (act), the Rogue's Shadow Shiv, Static Stiletto, Winter's Kiss and Permafrost Execution (they go
  // through act), and the mage's casts (castAct). NOT wired: the Umbral Shroud and the Abyssal Backstab (the blow comes from BEHIND: can a block stop it? the user's call), the monsters' attacks on a character, and a character
  // blocking (the characters have no block pack yet). The Elite's block has no strip of its own: it plays the Stone Guard's frames 2, 3, 4, 5, 3, 6 (the shield floats up in front of the face, the block frame is the face frame) at 14 fps.
  var BLOCK_REDUCE = 0.2, BLOCK_CAST_MS = 110, BLOCK_HOLD_MS = 380, BLOCK_DROP_MS = 380, BLOCK_VIS = 0.85;
  function blockAmount(amount) { return Math.max(1, Math.round(amount * (1 - BLOCK_REDUCE))); }
  function blockSpec() {                                                             // the monster's block: a pack played in an order (the Elite: the Stone Guard's frames), or its own strip
    var m = MONSTERS[monKey], b = m && m.block, pk;
    if (!b || !monEl) return null;
    pk = monPackOf(b.pack);
    if (!pk) return null;
    return { pk: pk, order: b.order || null, holds: b.holds || null, fps: b.fps || pk.fps, idx: b.idx || 0 };
  }
  function blockPre(sp) {                                                            // animation ms from the start of the pack to the moment the blow lands (40% into the block frame)
    var fm = 1000 / sp.fps, i, t = 0;
    for (i = 0; i < sp.idx; i++) t += fm * ((sp.holds && sp.holds[i]) || 1);
    return t + fm * 0.4 * ((sp.holds && sp.holds[sp.idx]) || 1);
  }
  function monBlock(myRun, delayMs, sp) {                                            // plays the block pack after delayMs (no movement: the guard goes up where it stands), then the idle
    return sleep(delayMs).then(function () {
      if (myRun !== run) return false;
      MON.blocking = true;                                                            // (the hit pack must not play for a blocked blow: monOnHp)
      var pk2 = Object.assign({}, sp.pk, { fps: sp.fps });
      return monPlayPack(pk2, myRun, null, sp.holds, sp.order).then(function (ok) { MON.blocking = false; if (ok && myRun === run) monStartIdle(); return ok; });
    });
  }
  function blockFx(T, sd, tHit) {                                                    // the mage's Barrier on the blocker, for a moment: the half circle on the side the blow comes from (sd: -1 left, +1 right), the ripple where it lands
    var mon = T === FIGHTERS.G && monEl, el = mon ? monEl : (T.unit.querySelector(".sheet") || T.unit), er = rectIn(el), w = mon ? er.width : uW.offsetWidth, sc = 1 / speed;
    var R = w * ((mon && MONSTERS[monKey].block && MONSTERS[monKey].block.r) || 0.5), bodyX = mon ? er.left + er.width * (259 / 512) : er.left + er.width / 2;      // (the Elite's base centre is x 259 of its 512 cell)
    var L = { t0: tHit, t1: tHit + (BLOCK_HOLD_MS + BLOCK_DROP_MS) * sc + 80, sc: sc, castMs: BLOCK_CAST_MS * sc, dropMs: BLOCK_DROP_MS * sc, endAt: tHit + BLOCK_HOLD_MS * sc, broken: false,
      rip: [{ t: tHit + 20 * sc, x: sd * 0.992, y: 0.125 }],
      geo: { sd: sd, r: R, cx: bodyX + sd * R * 0.16, gy: er.top + er.height - er.width * 42 / 512, vis: BLOCK_VIS },
      draw: function (ctx, now, l) { brDraw(ctx, now, l, false); }, drawBack: function (ctx, now, l) { brDraw(ctx, now, l, true); } };
    fxAdd(L);
  }
  monPrevLoop($("monPrevBlock"), async function (my) {                               // the Block preview: the guard goes up and a blow lands on the glass (nothing is dealt)
    var sp = blockSpec();
    if (!sp) return;
    var b = monBlock(my, 0, sp);
    await sleep(blockPre(sp));
    if (my !== run) return;
    blockFx(FIGHTERS.G, -FIGHTERS.W.dir, performance.now()); shake(3, 160);
    await b;
  });
