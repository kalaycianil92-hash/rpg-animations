  function kingPt(cx, cy) {                                                          // a point of the King's 512 cell (the sprite is mirrored on the stage) in stage px
    var r = rectIn(monEl), flip = uG.classList.contains("flip");
    return { x: r.left + (flip ? 1 - cx / 512 : cx / 512) * r.width, y: r.top + cy / 512 * r.height, w: r.width, h: r.height };
  }
  var KING_RUBY = [280, 176], KING_CROWN = [240, 150];
  function starGlint(ctx, x, y, r, a) {                                              // a four-point star of light
    if (a <= 0.01) return;
    glowDot(ctx, x, y, r * 1.2, "255,236,170", 0.6 * a);
    ctx.save(); ctx.globalCompositeOperation = "lighter"; ctx.lineCap = "round"; ctx.strokeStyle = "rgba(255,248,220," + (0.9 * a).toFixed(3) + ")"; ctx.lineWidth = Math.max(1, r * 0.09);
    ctx.beginPath(); ctx.moveTo(x - r, y); ctx.lineTo(x + r, y); ctx.moveTo(x, y - r * 1.25); ctx.lineTo(x, y + r * 1.25); ctx.stroke();
    ctx.lineWidth = Math.max(1, r * 0.05); ctx.beginPath(); ctx.moveTo(x - r * 0.55, y - r * 0.55); ctx.lineTo(x + r * 0.55, y + r * 0.55); ctx.moveTo(x - r * 0.55, y + r * 0.55); ctx.lineTo(x + r * 0.55, y - r * 0.55); ctx.stroke(); ctx.restore();
  }

    if (KA.ps.length > 40) KA.ps.splice(0, KA.ps.length - 40);
    for (i = KA.ps.length - 1; i >= 0; i--) {
      p = KA.ps[i]; p.age += dt; if (p.age >= p.life) { KA.ps.splice(i, 1); continue; }
      p.x += p.vx * dt + Math.sin(now * 0.0015 + i) * 0.006 * dt; p.y += p.vy * dt; f = p.age / p.life;
      glowDot(cf, p.x, p.y, p.size * 2.4, "255,226,140", 0.65 * Math.sin(f * 3.1416) * k);

    if (gt >= 0 && gt < 1) { gp = kingPt(KING_RUBY[0], KING_RUBY[1]); starGlint(cf, gp.x, gp.y, s * 0.045 * (0.5 + Math.sin(gt * 3.1416)), Math.sin(gt * 3.1416) * k); }

  function kingVignette(tS, tF, tF0, tE, aMax) {                                     // the stage's edges darken: from tS to full at tF, held until tF0, gone at tE (absolute times)
    fxAdd({ t0: tS, t1: tE, draw: function (ctx, now) {
      var a = aMax * clamp01((now - tS) / Math.max(1, tF - tS)) * (1 - clamp01((now - tF0) / Math.max(1, tE - tF0))), W = stage.clientWidth, H = stage.clientHeight, g;

      g.addColorStop(0, "rgba(12,8,0,0)"); g.addColorStop(1, "rgba(12,8,0," + a.toFixed(3) + ")");
      ctx.save(); ctx.fillStyle = g; ctx.fillRect(0, 0, W, H); ctx.restore();
    } });
  }

    var windMs = SLAM.windHolds.reduce(function (s0, h) { return s0 + h; }, 0) * fmR, t0 = performance.now(), tLeap = t0 + windMs, tImp = tLeap + (SLAM.crouchMs + SLAM.upMs + SLAM.downMs) / speed;

      if (!await bnDrive(myRun, SLAM.upMs / speed, function (s) { var q = 1 - (1 - s) * (1 - s); setT(reach * 0.5 * q, -H * q); })) return;
      showS(3);                                                                       // 4. the plunge, crown first
