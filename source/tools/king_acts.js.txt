    if (myRun !== run) return;
    var packOk = await ready;
    if (myRun !== run) return;
    if (packOk) monStartIdle();
    clearAnims(T.unit); clearAnims(stage);
    T.unit.querySelector(".sprite").getAnimations().forEach(function (x) { x.cancel(); });

        keyPose("death", 0); await sleep(150); keyPose("death", 1); await sleep(260); keyPose("death", 2);
